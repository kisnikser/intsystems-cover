\newcommand{\LectureCover}[3]{

                {\includegraphics[width=\paperwidth,height=\paperheight]{\BackgroundImage}};

                {\includegraphics[height=0.6\paperheight]{#2}};
            \node[anchor=west] at ($(current page.north west)+(0.8cm,-0.9cm)$)
                {\small\textcolor{alert}{\CourseName}};
            \node[anchor=west, text width=0.9\paperwidth] at
                ($(current page.west)+(0.8cm,1.0cm)$)

                {\huge Лекция~#1};

                {\small \LecturerLabel};
            \node[anchor=west] at ($(current page.south west)+(0.8cm,0.9cm)$)
                {\Large \textcolor{alert}{#3}};
        \end{tikzpicture}
    \end{frame}
}
